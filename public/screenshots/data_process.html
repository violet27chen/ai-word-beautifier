import pandas as pd
import os, glob

# 批量合并Excel文件 + 数据清洗 + 自动生成报表
def batch_process(folder):
    all_files = glob.glob(f"{folder}/*.xlsx")
    dfs = []
    for f in all_files:
        df = pd.read_excel(f)
        df["来源"] = os.path.basename(f)
        dfs.append(df)
    merged = pd.concat(dfs, ignore_index=True)
    # 数据清洗
    merged = merged.drop_duplicates()
    merged = merged.dropna(subset=["姓名", "电话"])
    merged["电话"] = merged["电话"].astype(str).str.replace(r"\D", "", regex=True)
    # 生成汇总报表
    summary = merged.groupby("来源").agg(
        计数=("姓名", "count"),
        唯一电话=("电话", "nunique")
    )
    with pd.ExcelWriter("汇总报表.xlsx") as writer:
        merged.to_excel(writer, sheet_name="明细", index=False)
        summary.to_excel(writer, sheet_name="汇总")
    return merged, summary
